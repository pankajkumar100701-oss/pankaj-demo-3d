anomalies.py
import strata

db = strata.connect("prod-eu")

# every checkout, bucketed by minute
peaks = db.query("""
  SELECT minute, region, count(*) AS n
  FROM checkout.events
  WINDOW 1 HOUR
""").detect(sensitivity=0.9)

for p in peaks:
    print(p.region, p.score)  # eu-west 4.7σ
